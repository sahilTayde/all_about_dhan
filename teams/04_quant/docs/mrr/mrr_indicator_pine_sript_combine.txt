//@version=5
indicator("Option Scalper Ultimate (MRR + Fib + Sweeps)", overlay=true)

// --- Inputs ---
mrr_length   = input.int(20, title="MRR Base Period")
ema_fast_len = input.int(9, title="Fast EMA")
ema_slow_len = input.int(21, title="Slow EMA")
rsi_len      = input.int(14, title="RSI Period")
vol_mult     = input.float(1.5, title="Volume Spike Multiplier")
pivot_len    = input.int(5, title="Liquidity Pivot Lookback")

// --- Calculations ---
mrr          = ta.vwma(close, mrr_length)
vwap_val     = ta.vwap(hlc3)
ema_fast     = ta.ema(close, ema_fast_len)
ema_slow     = ta.ema(close, ema_slow_len)
rsi_val      = ta.rsi(close, rsi_len)
vol_spike    = volume > (ta.sma(volume, 20) * vol_mult)

// --- Pivots & Sweeps ---
ph = ta.pivothigh(high, pivot_len, pivot_len)
pl = ta.pivotlow(low, pivot_len, pivot_len)
var float res = na
var float sup = na
if not na(ph) 
    res := ph
if not na(pl) 
    sup := pl

bull_sweep = low < sup and close > sup

// --- Fibonacci ---
hi_50 = ta.highest(high, 50)
lo_50 = ta.lowest(low, 50)
fib_618 = hi_50 - ((hi_50 - lo_50) * 0.618)

// --- Strategy Triggers ---
// 1. Liquidity Sweep Signal
sig_sweep = bull_sweep and close > mrr

// 2. Breakout Signal
sig_breakout = (close > mrr) and (close > vwap_val) and vol_spike

// 3. Fib Trend Continuation Signal
sig_fib = (low <= fib_618) and (close > ema_fast) and (ema_fast > ema_slow) and (rsi_val > 50)

// --- Plots ---
plot(mrr, "MRR Line", color=color.blue, linewidth=2)
plot(vwap_val, "VWAP", color=color.purple, linewidth=1)
plot(ema_fast, "Fast EMA", color=color.green)
plot(ema_slow, "Slow EMA", color=color.red)
plot(fib_618, "Fib 0.618", color=color.gold, style=plot.style_dashed)

// --- Markers ---
plotshape(sig_sweep, "Sweep Entry", shape.labelup, location.belowbar, color.cyan, text="SWEEP", textcolor=color.black)
plotshape(sig_breakout, "Breakout Entry", shape.labelup, location.belowbar, color.green, text="BREAKOUT", textcolor=color.white)
plotshape(sig_fib, "Fib Entry", shape.labelup, location.belowbar, color.orange, text="FIB", textcolor=color.white)
